    <script>
        document.addEventListener('DOMContentLoaded', function() {
            loadCountriesList();
            loadRecentActivity();
            setupMapInteractions();
        });

        function loadCountriesList() {
            const countries = [
                { name: 'Ethiopia', count: 234 },
                { name: 'Kenya', count: 156 },
                { name: 'Nigeria', count: 189 },
                { name: 'Ghana', count: 98 },
                { name: 'South Africa', count: 145 },
                { name: 'India', count: 312 },
                { name: 'USA', count: 423 },
                { name: 'UK', count: 201 }
            ];
            
            const grid = document.getElementById('countriesGrid');
            grid.innerHTML = '';
            
            countries.sort((a, b) => b.count - a.count).forEach(c => {
                const item = document.createElement('div');
                item.className = 'country-item';
                item.innerHTML = `<span class="country-name">${c.name}</span><span class="country-count">${c.count}</span>`;
                item.addEventListener('click', () => alert(`🌍 ${c.name}\n\n${c.count} peace voices`));
                grid.appendChild(item);
            });
        }

        function loadRecentActivity() {
            const activities = [
                { time: '2 minutes ago', from: 'Ethiopia', to: 'Kenya', word: 'Love' },
                { time: '15 minutes ago', from: 'India', to: 'USA', word: 'Hope' },
                { time: '1 hour ago', from: 'Nigeria', to: 'Ghana', word: 'Unity' }
            ];
            
            const feed = document.getElementById('activityFeed');
            feed.innerHTML = '';
            
            activities.forEach(a => {
                const item = document.createElement('div');
                item.className = 'activity-item';
                item.innerHTML = `
                    <div class="activity-time">${a.time}</div>
                    <div class="activity-text"><strong>${a.from}</strong> → <strong>${a.to}</strong><br>"${a.word}"</div>
                `;
                feed.appendChild(item);
            });
        }

        function setupMapInteractions() {
            document.querySelectorAll('.country-dot').forEach(dot => {
                dot.addEventListener('click', function() {
                    alert(`🗺️ ${this.getAttribute('data-country')}\n\nPeace voices from this country coming soon.`);
                });
            });
        }
