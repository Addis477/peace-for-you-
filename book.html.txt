    <script>
        // Book of Lights specific JavaScript
        document.addEventListener('DOMContentLoaded', function() {
            loadFeaturedVoice();
            loadAllLights();
            setupBookListeners();
        });

        function loadFeaturedVoice() {
            const featuredContainer = document.getElementById('featuredVoice');
            featuredContainer.innerHTML = `
                <div class="featured-label">✨ VOICE OF THE DAY</div>
                <div class="featured-name">Maria Tadesse</div>
                <div class="featured-country">Ethiopia</div>
                <div class="featured-word">"Love"</div>
                <div class="featured-message">"Love is the only force that grows when shared."</div>
            `;
        }

        function loadAllLights() {
            const lightsGrid = document.getElementById('lightsGrid');
            
            const samples = [
                { name: 'Maria Tadesse', country: 'Ethiopia', word: 'Love', type: 'personal' },
                { name: 'John Omondi', country: 'Kenya', word: 'Hope', type: 'personal' },
                { name: 'David & Sarah', country: 'Ghana', word: 'Together', type: 'couple' },
                { name: 'Abebe Family', country: 'Ethiopia', word: 'Unity', type: 'family' },
                { name: 'Peace Builders', country: 'Nigeria', word: 'Action', type: 'group' }
            ];
            
            lightsGrid.innerHTML = '';
            samples.forEach(sample => {
                const card = createLightCard(sample);
                lightsGrid.appendChild(card);
            });
        }

        function createLightCard(item) {
            const card = document.createElement('div');
            card.className = 'light-card';
            card.innerHTML = `
                <div class="light-card-header">
                    <span class="light-name">${item.name}</span>
                    <span class="light-country">${item.country}</span>
                </div>
                <div class="light-word">"${item.word}"</div>
                <div class="light-footer">
                    <span class="light-type">${item.type}</span>
                </div>
            `;
            return card;
        }

        function setupBookListeners() {
            document.getElementById('searchBtn').addEventListener('click', function() {
                alert('🔍 Search coming soon with Supabase integration.');
            });
            
            document.querySelectorAll('.country-tag').forEach(tag => {
                tag.addEventListener('click', function() {
                    document.querySelectorAll('.country-tag').forEach(t => t.classList.remove('active'));
                    this.classList.add('active');
                });
            });
            
            document.getElementById('loadMoreBtn').addEventListener('click', function() {
                alert('📖 More voices coming soon.');
            });
        }
    </script>
