    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const fullSky = document.getElementById('fullSky');
            if (fullSky) fullSky.classList.add('sky-canvas');
            
            const timeDisplay = document.getElementById('currentTime');
            if (window.skyAPI) {
                const times = {
                    'dawn': 'Dawn 🌅',
                    'morning': 'Morning ☀️',
                    'afternoon': 'Afternoon 🌤️',
                    'evening': 'Evening 🌆',
                    'sunset': 'Sunset 🌇',
                    'night': 'Night 🌙'
                };
                const current = window.skyAPI.getTimeOfDay();
                if (timeDisplay) timeDisplay.textContent = times[current] || 'Live Sky';
                
                setInterval(() => {
                    const updated = window.skyAPI.getTimeOfDay();
                    timeDisplay.textContent = times[updated] || 'Live Sky';
                }, 60000);
            }
        });
    </script>
